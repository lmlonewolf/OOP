cmake_minimum_required(VERSION 3.16)
project(text_index LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(text_index main.cpp Tests.cpp Document.cpp TextIndex.cpp)
file(COPY documents test_data DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
set_property(TARGET text_index PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(text_index PRIVATE /W4 /utf-8)
else()
    target_compile_options(text_index PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME model_tests COMMAND text_index --test)
