cmake_minimum_required(VERSION 3.16)
project(schedule LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(schedule main.cpp Tests.cpp DateTime.cpp Lessons.cpp Constraints.cpp Schedule.cpp)
if(MSVC)
    target_compile_options(schedule PRIVATE /W4 /utf-8)
else()
    target_compile_options(schedule PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME model_tests COMMAND schedule --test)
